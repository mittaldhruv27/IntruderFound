# paths, absolute paths, parent directory traversal (..), and root (/).
# ------------------------------------------------------------------------------

# Feature: Change directory to parent directory (/home)

# Feature: Navigate to root filesystem (/)

ls -la

# Feature: Absolute path navigation directly into system log directory

# Feature: Return to user home directory using ~ shortcut

# ------------------------------------------------------------------------------
# 3. VFS FILE CREATION, WRITING & MANIPULATION
# Displays: In-memory file creation, content inspection, file copying, renaming,
# and deletion without touching host macOS/Linux disk files.
# ------------------------------------------------------------------------------

# Feature: Create a new custom directory in the virtual filesystem
mkdir test_vault
ls -la

# Feature: Enter the newly created directory
cd test_vault
pwd

# Feature: Create an empty file with dynamic creation timestamp
touch scan_results.txt
ls -la

# Feature: Write text content into virtual file using echo redirection
echo "Host 192.168.1.50 port 22 open" > scan_results.txt

# Feature: Read and display file contents from virtual inode memory
cat scan_results.txt

# Feature: Copy file to another target path in VFS
cp scan_results.txt backup_scan.txt
ls -la

# Feature: Rename/move file within the virtual filesystem
mv backup_scan.txt final_report.txt
ls -la

# Feature: Read the renamed file to verify data integrity
cat final_report.txt

# Feature: Delete a file from VFS and verify removal
rm final_report.txt
ls -la

# Feature: Return back to user home directory
cd /home/user123
